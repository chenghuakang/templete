cmake_minimum_required(VERSION 3.20)
project(satpred LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
foreach(cfg DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
 set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${cfg} ${CMAKE_BINARY_DIR}/bin)
endforeach()
set(SATPRED_CUDA OFF CACHE STRING "OFF AUTO ON")
option(SATPRED_TESTS "Build smoke tests" OFF)
option(SATPRED_WERROR "Warnings as errors" ON)
set(SATPRED_CUDA_ENABLED OFF)
if(NOT SATPRED_CUDA STREQUAL "OFF")
 find_program(NVCC_EXECUTABLE nvcc)
 if(NVCC_EXECUTABLE)
  enable_language(CUDA)
  set(SATPRED_CUDA_ENABLED ON)
 elseif(SATPRED_CUDA STREQUAL "ON")
  message(FATAL_ERROR "SATPRED_CUDA=ON but nvcc was not found")
 endif()
endif()
find_package(OpenMP)
add_library(satcore STATIC src/core.cpp)
if(SATPRED_CUDA_ENABLED)
 target_sources(satcore PRIVATE src/cuda_candidates.cu)
 target_compile_definitions(satcore PRIVATE SP_CUDA=1)
 if(NOT MSVC)
  target_compile_options(satcore PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
 endif()
endif()
target_include_directories(satcore PUBLIC include)
if(MSVC)
 target_compile_options(satcore PRIVATE /O2 /EHsc /fp:strict /utf-8)
 if(SATPRED_WERROR)
 target_compile_options(satcore PRIVATE /WX)
 endif()
 target_compile_definitions(satcore PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
else()
 target_compile_options(satcore PRIVATE -O2 -Wall -Wextra -ffp-contract=off -fno-fast-math)
 if(SATPRED_WERROR)
 target_compile_options(satcore PRIVATE -Werror)
 endif()
endif()
if(OpenMP_CXX_FOUND)
 target_link_libraries(satcore PUBLIC OpenMP::OpenMP_CXX)
endif()
foreach(app satpred satnet satread netread satdemo orbit_engine)
 add_executable(${app} apps/main.cpp)
 target_compile_definitions(${app} PRIVATE APP_NAME="${app}")
 target_link_libraries(${app} PRIVATE satcore)
 install(TARGETS ${app} RUNTIME DESTINATION bin)
endforeach()
file(COPY examples DESTINATION ${CMAKE_BINARY_DIR}/bin)
install(DIRECTORY examples DESTINATION bin)
if(SATPRED_TESTS)
 enable_testing()
 add_test(NAME selftest COMMAND satpred --selftest)
endif()
