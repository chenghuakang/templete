cmake_minimum_required(VERSION 3.20)
project(UnifiedSatNet LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
foreach(cfg DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
 set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${cfg} ${CMAKE_BINARY_DIR}/bin)
endforeach()
option(UNIFIED_TESTS "Build tests" ON)
add_subdirectory(satpred)
# simulation.cpp is one translation unit: it includes src/options.inc, src/runtime_io.inc and src/handover.inc
# because include/hsn/reader_internal.hpp includes satpred/src/core.cpp directly.
add_library(hsncore STATIC src/simulation.cpp)
target_include_directories(hsncore PUBLIC include satpred/include)
add_executable(hypersim src/main.cpp)
target_link_libraries(hypersim PRIVATE hsncore)
add_executable(hsn-traceroute src/trace_main.cpp)
target_include_directories(hsn-traceroute PRIVATE include satpred/include)
foreach(target hsncore hsn-traceroute)
 if(MSVC)
  target_compile_options(${target} PRIVATE /O2 /EHsc /fp:strict /utf-8)
  target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 else()
  target_compile_options(${target} PRIVATE -O2 -Wall -Wextra -Wno-deprecated-declarations -ffp-contract=off -fno-fast-math)
 endif()
endforeach()
install(TARGETS hypersim hsn-traceroute RUNTIME DESTINATION bin)
if(UNIFIED_TESTS)
 enable_testing()
 add_executable(test_options tests/test_options.cpp)
 target_include_directories(test_options PRIVATE include satpred/include)
 add_test(NAME options_and_event_order COMMAND test_options)
 add_executable(test_traceroute tests/test_traceroute.cpp)
 target_include_directories(test_traceroute PRIVATE include)
 add_test(NAME traceroute_correctness COMMAND test_traceroute)
 add_test(NAME common_hash_time COMMAND hypersim --selftest)
 find_package(Python3 COMPONENTS Interpreter QUIET)
 if(Python3_Interpreter_FOUND)
  add_test(NAME unified_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/regression.py $<TARGET_FILE_DIR:hypersim> ${CMAKE_BINARY_DIR}/test_runs)
  set_tests_properties(unified_integration PROPERTIES TIMEOUT 900)
  add_test(NAME default_flow_clean_env COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/default_flow.py $<TARGET_FILE_DIR:hypersim> ${CMAKE_BINARY_DIR}/test_runs)
  set_tests_properties(default_flow_clean_env PROPERTIES TIMEOUT 1800)
  add_test(NAME upstream_regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/satpred/tests/regression.py $<TARGET_FILE_DIR:hypersim>)
  set_tests_properties(upstream_regression PROPERTIES TIMEOUT 900)
   add_test(NAME startup_jitter COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/startup_jitter.py $<TARGET_FILE_DIR:hypersim> ${CMAKE_BINARY_DIR}/test_runs)
   set_tests_properties(startup_jitter PROPERTIES TIMEOUT 900)

 endif()
endif()
